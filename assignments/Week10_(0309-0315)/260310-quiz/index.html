(260310) DFS & BFS

1. DFS와 BFS — 백준 1260 .js

[문제 주소]

https://www.acmicpc.net/problem/1260

[사고방식 및 풀이 과정]

1. 정렬이 핵심: 동일한 거리에 노드가 여러 개일 때 작은 번호를 먼저 가야 하므로 오름차순 정렬이 필수다.

2. DFS: 한 놈만 끝까지 패는 전략 (재귀 호출 활용).

3. BFS: 주변부터 넓게 훑는 전략 (Queue 자료구조 활용).

Node.js
const fs = require('fs');
const input = fs.readFileSync('/dev/stdin').toString().trim().split('\n');

// N: 정점 개수, M: 간선 개수, V: 시작 정점
const [N, M, V] = input[0].split(' ').map(Number);
const graph = Array.from({ length: N + 1 }, () => []);

// 인접 리스트 빌드 (양방향 연결)
for (let i = 1; i <= M; i++) {
    const [u, v] = input[i].split(' ').map(Number);
    graph[u].push(v);
    graph[v].push(u);
}

// 번호가 작은 노드부터 방문하기 위해 오름차순 정렬
graph.forEach(arr => arr.sort((a, b) => a - b));

// --- DFS 구현 (재귀) ---
const dfsResult = [];
const visitedD = new Array(N + 1).fill(false);

function dfs(v) {
    visitedD[v] = true; // 방문 도장 쾅!
    dfsResult.push(v);
    for (const next of graph[v]) {
        if (!visitedD[next]) dfs(next); // 미방문 노드로 깊게 파고들기
    }
}

// --- BFS 구현 (Queue) ---
const bfsResult = [];
const visitedB = new Array(N + 1).fill(false);

function bfs(start) {
    const q = [start];
    visitedB[start] = true;
    while (q.length) {
        const v = q.shift(); // 큐 앞에서 노드 꺼내기
        bfsResult.push(v);
        for (const next of graph[v]) {
            if (!visitedB[next]) {
                visitedB[next] = true;
                q.push(next); // 주변 노드 큐에 예약하기
            }
        }
    }
}

dfs(V); bfs(V);
console.log(dfsResult.join(' '));
console.log(bfsResult.join(' '));
2. 단지번호붙이기 — 백준 2667 .js

[문제 주소]

https://www.acmicpc.net/problem/2667

[사고방식 및 풀이 과정]

1. 전수 조사: 지도의 모든 칸을 뒤지다가 1(집)을 만나면 그 단지를 박살(?)낸다.

2. Flood Fill: 1인 곳을 방문하면 다시는 못 오게 0으로 지워버린다.

Node.js
const fs = require('fs');
const input = fs.readFileSync('/dev/stdin').toString().trim().split('\n');
const N = Number(input[0]);
// 문자열로 들어오는 지도를 2차원 숫자 배열로 변환
const map = input.slice(1).map(l => l.split('').map(Number));

// 상하좌우 이동을 위한 방향 벡터
const dr = [-1, 1, 0, 0], dc = [0, 0, -1, 1];
const complexes = [];

function dfs(r, c) {
    map[r][c] = 0; // 방문한 집은 0으로 만들어 중복 카운트 방지
    let count = 1;
    for (let i = 0; i < 4; i++) {
        const nr = r + dr[i], nc = c + dc[i];
        // 지도 범위 내에 있고 아직 방문하지 않은 집(1)인 경우
        if (nr >= 0 && nr < N && nc >= 0 && nc < N && map[nr][nc] === 1) {
            count += dfs(nr, nc);
        }
    }
    return count;
}

// 지도를 순회하며 단지 찾기
for (let i = 0; i < N; i++) {
    for (let j = 0; j < N; j++) {
        if (map[i][j] === 1) {
            complexes.push(dfs(i, j)); // 새 단지 발견! 크기 측정 시작
        }
    }
}

// 오름차순 정렬 후 출력 규격에 맞춰 출력
complexes.sort((a, b) => a - b);
console.log(complexes.length + '\n' + complexes.join('\n'));
3. 유기농 배추 — 백준 1012 .js

[문제 주소]

https://www.acmicpc.net/problem/1012

[사고방식 및 풀이 과정]

1. 연결 요소 개수: 배추들이 모여있는 덩어리가 몇 개인지를 찾는 문제다.

2. 다중 테스트 케이스: T만큼 반복문을 돌리며 매번 새로운 필드를 생성해야 함에 주의한다.

Node.js
const fs = require('fs');
const input = fs.readFileSync('/dev/stdin').toString().trim().split('\n');
let line = 0;
const T = Number(input[line++]); // 테스트 케이스 개수

while (line < input.length) {
    // M: 가로, N: 세로, K: 배추 개수
    const [M, N, K] = input[line++].split(' ').map(Number);
    const field = Array.from({ length: N }, () => Array(M).fill(0));

    // 배추 심기 (좌표 Y, X 주의)
    for (let i = 0; i < K; i++) {
        const [x, y] = input[line++].split(' ').map(Number);
        field[y][x] = 1;
    }

    let worms = 0;
    // 전체 밭을 돌며 지렁이가 필요한 배추 덩어리 탐색
    for (let i = 0; i < N; i++) {
        for (let j = 0; j < M; j++) {
            if (field[i][j] === 1) {
                worms++; // 지렁이 1마리 추가
                bfs(i, j, field, N, M); // 연결된 모든 배추 방문 완료 처리
            }
        }
    }
    console.log(worms);
}

function bfs(r, c, field, N, M) {
    const q = [[r, c]];
    field[r][c] = 0; // 방문 처리
    const dr = [-1, 1, 0, 0], dc = [0, 0, -1, 1];
    while (q.length) {
        const [currR, currC] = q.shift();
        for (let i = 0; i < 4; i++) {
            const nr = currR + dr[i], nc = currC + dc[i];
            if (nr >= 0 && nr < N && nc >= 0 && nc < M && field[nr][nc] === 1) {
                field[nr][nc] = 0;
                q.push([nr, nc]);
            }
        }
    }
}
